    import chromadb
    from chromadb.utils.batch_utils import create_batches

    def insert_collection(
        client: chromadb.ClientAPI,
        collection: chromadb.Collection,
        documents: list[str],
        metadatas: list[dict],
        ids: list[str],
        collection_name: str,
    ) -> int:
        """Insert records into a ChromaDB collection with automatic batching and progress logging.

        Uses collection.upsert() for idempotent behavior (D-10 skip + re-run safety).
        Uses chromadb.utils.batch_utils.create_batches() for automatic batch size calculation.

        Args:
            client: ChromaDB client (needed by create_batches for max_batch_size)
            collection: Target ChromaDB collection
            documents: Document texts to embed
            metadatas: Metadata dicts (must be pre-flattened)
            ids: Unique document IDs
            collection_name: Name for progress logging

        Returns:
            Number of records inserted
        """
        if not ids:
            logger.info(f"{collection_name}: 0 records, skipping")
            return 0

        batches = create_batches(
            api=client,
            ids=ids,
            documents=documents,
            metadatas=metadatas,
        )
        total_batches = len(batches)
        for i, batch in enumerate(batches, 1):
            batch_ids, batch_embeddings, batch_metadatas, batch_documents = batch
            collection.upsert(
                ids=batch_ids,
                documents=batch_documents,
                metadatas=batch_metadatas,
            )
            logger.info(
                f"{collection_name}: batch {i}/{total_batches} "
                f"({len(batch_ids)} records)"
            )
        logger.info(f"{collection_name}: {len(ids)} records inserted")
        return len(ids)
    ```

        """Load multiple talk bundles into ChromaDB collections in dependency order.

        Invalid talks (validation.is_valid=False) are skipped per D-10.

        Args:
            client: ChromaDB client (PersistentClient for production, EphemeralClient for tests)
            bundles: List of (metadata, paragraphs, citations, footnotes, validation) tuples

        Returns:
            Dict with keys:
                - "inserted_counts": {"conference_talks": N, "conference_talk_paragraphs": N, ...}
                - "skipped_talks": [talk_id, ...]
                - "total_talks": int
                - "valid_talks": int
        """
    ```
