**Step 2 — Rewrite `auth/oauth.py` verify_token method:**

Replace the current `verify_token` method with kid-based routing. The new logic:

```python
def verify_token(self, token: str) -> dict | None:
    # JWE compact has 5 segments (4 dots) — route to /userinfo
    if token.count(".") == 4:
        logger.debug("JWE token detected, routing to /userinfo validation")
        return self._validate_via_userinfo(token)

    try:
        header = jwt.get_unverified_header(token)
    except jwt.PyJWTError:
        return self._validate_via_userinfo(token)

    if header.get("kid"):
        return self._verify_asymmetric(token, header["alg"])
    else:
        return self._validate_via_userinfo(token)
```

Remove these class attributes entirely:
- `ASYMMETRIC_ALGORITHMS` list
- `SYMMETRIC_ALGORITHMS` list

Remove all references to `self.ASYMMETRIC_ALGORITHMS` and `self.SYMMETRIC_ALGORITHMS` from `verify_token`.

Keep `_validate_via_userinfo` and `_verify_asymmetric` methods unchanged.

Update the module docstring to reflect kid-based routing instead of alg-based routing:
- "Inspects the JWT header for `kid` claim presence" instead of "Inspects the JWT header to determine the signing algorithm"
- Remove references to SYMMETRIC_ALGORITHMS and ASYMMETRIC_ALGORITHMS lists

```python
def test_verify_rs256_without_kid_routes_to_userinfo(rsa_private_key):
    """RS256 token WITHOUT kid header routes to /userinfo (kid-based routing)."""
    verifier = _make_verifier()
    # Encode RS256 token without kid header
    token = jwt.encode(
        {"sub": "user-no-kid", "aud": FAKE_AUDIENCE, "iss": FAKE_ISSUER, "exp": int(time.time()) + 300},
        rsa_private_key,
        algorithm="RS256",
        # Deliberately NO kid header
    )
    mock_resp = _mock_userinfo_response(200, {"sub": "user-no-kid"})
    with patch("auth.oauth.httpx.Client") as MockClient:
        MockClient.return_value.__enter__.return_value.get.return_value = mock_resp
        result = verifier.verify_token(token)
    assert result is not None
    assert result["sub"] == "user-no-kid"


def test_verify_hs256_with_kid_routes_to_jwks():
    """HS256 token WITH kid header routes to JWKS (kid-based routing)."""
    verifier = _make_verifier()
    # HS256 with kid — kid presence means JWKS path, which will fail
    # (no real JWKS endpoint) and return None
    token = jwt.encode(
        {"sub": "user-hs-kid", "aud": FAKE_AUDIENCE, "iss": FAKE_ISSUER, "exp": int(time.time()) + 300},
        "any-secret",
        algorithm="HS256",
        headers={"kid": "unexpected-kid"},
    )
    # JWKS lookup will fail (mock raises), so verify_token returns None
    verifier._jwks_client.get_signing_key_from_jwt.side_effect = Exception("key not found")
    result = verifier.verify_token(token)
    assert result is None
```
