Full file content to write (verbatim — all values verified against official docs and RESEARCH.md):

```yaml
services:
  chromadb:
    image: chromadb/chroma:1.5.7
    container_name: chromadb-benchmark
    restart: unless-stopped
    ports:
      - "127.0.0.1:8000:8000"
    volumes:
      - chromadb_data:/data
    networks:
      - benchmark
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/api/v1/heartbeat"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 30s

  weaviate:
    image: cr.weaviate.io/semitechnologies/weaviate:1.37.1
    container_name: weaviate-benchmark
    restart: unless-stopped
    ports:
      - "127.0.0.1:8080:8080"
      - "127.0.0.1:50051:50051"
    environment:
      AUTHENTICATION_ANONYMOUS_ACCESS_ENABLED: "true"
      PERSISTENCE_DATA_PATH: "/var/lib/weaviate"
      DEFAULT_VECTORIZER_MODULE: "none"
      CLUSTER_HOSTNAME: "weaviate-benchmark"
    volumes:
      - weaviate_data:/var/lib/weaviate
    networks:
      - benchmark
    healthcheck:
      test: ["CMD", "wget", "--no-verbose", "--tries=1", "--spider",
             "http://localhost:8080/v1/.well-known/ready"]
      interval: 10s
      timeout: 30s
      retries: 5
      start_period: 30s

  qdrant:
    image: qdrant/qdrant:v1.17.1
    container_name: qdrant-benchmark
    restart: unless-stopped
    ports:
      - "127.0.0.1:6333:6333"
      - "127.0.0.1:6334:6334"
    volumes:
      - qdrant_data:/qdrant/storage
    networks:
      - benchmark
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:6333/healthz"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 30s

networks:
  benchmark:
    name: benchmark
    driver: bridge

volumes:
  chromadb_data:
  weaviate_data:
  qdrant_data:
```

Write this exactly. Do NOT add Ollama — Ollama runs on the host (SPEC out-of-scope). Do NOT join the proxy network — benchmark network is isolated per SPEC.

| Boundary | Description |
|----------|-------------|
| host → vector DB containers | Ports published to localhost only (127.0.0.1) |
| benchmark network | Isolated from production proxy network |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-20-04-01 | Information Disclosure | Vector DB ports exposed on non-loopback | mitigate | All ports use "127.0.0.1:PORT:PORT" binding — only localhost can connect; prevents remote access on shared/multi-user machines |
| T-20-04-02 | Spoofing | No authentication on benchmark DBs | accept | SPEC explicitly scopes Phase 20 to local dev, anonymous access only; auth not in scope until production migration |
| T-20-04-03 | Elevation of Privilege | benchmark network joining proxy network | mitigate | benchmark network is isolated (name: benchmark, driver: bridge); no cross-network references; confirmed by absence of "proxy" in compose file |
