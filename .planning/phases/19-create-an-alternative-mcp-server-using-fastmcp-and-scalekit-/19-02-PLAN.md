**Part A: Create `auth/scalekit_verifier.py`**

New file implementing `ScalekitTokenVerifier(TokenVerifier)`:

```python
"""
ScaleKit JWT verification for FastMCP.

Wraps ScalekitClient.validate_token() as a FastMCP TokenVerifier subclass.
ScaleKit SDK is synchronous -- all validation calls run in a thread via
anyio.to_thread.run_sync() to avoid blocking the event loop.
"""
import logging
import anyio
from scalekit import ScalekitClient
from fastmcp.server.auth import TokenVerifier, AccessToken

logger = logging.getLogger(__name__)


class ScalekitTokenVerifier(TokenVerifier):
    """Verify ScaleKit-issued JWTs for use as FastMCP auth.

    ScalekitClient is constructed eagerly (in __init__) because its
    constructor calls __authenticate_client() synchronously. This must
    happen before the event loop starts (at module/startup level).
    """

    def __init__(self, env_url: str, client_id: str, client_secret: str):
        super().__init__()
        self._client = ScalekitClient(
            env_url=env_url,
            client_id=client_id,
            client_secret=client_secret,
        )

    async def verify_token(self, token: str) -> AccessToken | None:
        """Verify a ScaleKit JWT. Returns AccessToken or None on failure."""
        try:
            payload = await anyio.to_thread.run_sync(
                lambda: self._client.validate_token(token)
            )
        except Exception:
            logger.debug("ScaleKit token validation failed")
            return None

        scopes_raw = payload.get("scope", "")
        scopes = scopes_raw.split() if isinstance(scopes_raw, str) else []

        return AccessToken(
            token=token,
            client_id=payload.get("client_id") or payload.get("azp") or payload.get("sub", ""),
            scopes=scopes,
            expires_at=payload.get("exp"),
            claims=payload,
        )
```

Do NOT log the token value or client secret. Do NOT catch specific exception types -- catch `Exception` broadly and return `None` (graceful 401). This handles key rotation (KeyError), expired tokens, malformed tokens, and network errors uniformly.

class TestScalekitVerifier:
    """Unit tests for ScalekitTokenVerifier."""

    def test_valid_token_returns_access_token(self):
        """verify_token returns AccessToken when ScaleKit validates successfully."""
        from auth.scalekit_verifier import ScalekitTokenVerifier
        from fastmcp.server.auth import AccessToken

        verifier = ScalekitTokenVerifier.__new__(ScalekitTokenVerifier)
        verifier._client = MagicMock()
        verifier._client.validate_token.return_value = {
            "sub": "user123",
            "client_id": "client_abc",
            "scope": "read write",
            "exp": 9999999999,
        }

        result = asyncio.run(verifier.verify_token("valid-token"))

        assert isinstance(result, AccessToken)
        assert result.client_id == "client_abc"
        assert result.scopes == ["read", "write"]
        assert result.expires_at == 9999999999
        assert result.claims["sub"] == "user123"

    def test_invalid_token_returns_none(self):
        """verify_token returns None when ScaleKit raises an exception."""
        from auth.scalekit_verifier import ScalekitTokenVerifier

        verifier = ScalekitTokenVerifier.__new__(ScalekitTokenVerifier)
        verifier._client = MagicMock()
        verifier._client.validate_token.side_effect = Exception("invalid token")

        result = asyncio.run(verifier.verify_token("bad-token"))

        assert result is None

    def test_key_error_returns_none(self):
        """verify_token returns None on KeyError (unknown kid / key rotation)."""
        from auth.scalekit_verifier import ScalekitTokenVerifier

        verifier = ScalekitTokenVerifier.__new__(ScalekitTokenVerifier)
        verifier._client = MagicMock()
        verifier._client.validate_token.side_effect = KeyError("unknown_kid")

        result = asyncio.run(verifier.verify_token("rotated-key-token"))

        assert result is None

    def test_missing_scope_defaults_empty(self):
        """verify_token handles missing scope claim gracefully."""
        from auth.scalekit_verifier import ScalekitTokenVerifier
        from fastmcp.server.auth import AccessToken

        verifier = ScalekitTokenVerifier.__new__(ScalekitTokenVerifier)
        verifier._client = MagicMock()
        verifier._client.validate_token.return_value = {"sub": "user1"}

        result = asyncio.run(verifier.verify_token("token-no-scope"))

        assert isinstance(result, AccessToken)
        assert result.scopes == []

    def test_azp_fallback_for_client_id(self):
        """verify_token uses azp when client_id is absent in payload."""
        from auth.scalekit_verifier import ScalekitTokenVerifier
        from fastmcp.server.auth import AccessToken

        verifier = ScalekitTokenVerifier.__new__(ScalekitTokenVerifier)
        verifier._client = MagicMock()
        verifier._client.validate_token.return_value = {"sub": "user1", "azp": "azp_client"}

        result = asyncio.run(verifier.verify_token("token-azp"))

        assert isinstance(result, AccessToken)
        assert result.client_id == "azp_client"
