import asyncio
from mcp.server.auth.provider import TokenVerifier, AccessToken
from scalekit import ScalekitClient

class ScalekitTokenVerifier:
    def __init__(self, client: ScalekitClient, audience: str | None = None):
        self._client = client
        self._audience = audience

    async def verify_token(self, token: str) -> AccessToken | None:
        try:
            payload = await asyncio.to_thread(
                self._client.validate_token, token, audience=self._audience
            )
        except Exception:
            return None
        return AccessToken(
            token=token,
            client_id=payload.get("sub", ""),
            scopes=[],
            expires_at=payload.get("exp"),
        )
```

class ScalekitTokenVerifier:
    """TokenVerifier implementation backed by ScaleKit JWKS validation.
    
    ScalekitClient.validate_token() is synchronous (uses requests.get),
    so we wrap it in asyncio.to_thread() to avoid blocking the event loop.
    """
    def __init__(self, client, audience: str | None = None):
        self._client = client
        self._audience = audience

    async def verify_token(self, token: str):
        from mcp.server.auth.provider import AccessToken
        try:
            payload = await asyncio.to_thread(
                self._client.validate_token, token, audience=self._audience
            )
        except Exception:
            return None
        return AccessToken(
            token=token,
            client_id=payload.get("sub", ""),
            scopes=[],
            expires_at=payload.get("exp"),
        )
```

class TestScalekitTokenVerifier:
    """Unit tests for ScalekitTokenVerifier."""

    @pytest.mark.asyncio
    async def test_valid_token_returns_access_token(self):
        """verify_token returns AccessToken with sub and exp from payload."""
        mock_client = MagicMock()
        mock_client.validate_token.return_value = {
            "sub": "client-abc",
            "exp": 9999999999,
            "iss": "https://test.scalekit.com",
        }
        verifier = ScalekitTokenVerifier(mock_client, audience="test-audience")
        result = await verifier.verify_token("valid-jwt-token")
        assert result is not None
        assert result.token == "valid-jwt-token"
        assert result.client_id == "client-abc"
        assert result.expires_at == 9999999999
        mock_client.validate_token.assert_called_once_with(
            "valid-jwt-token", audience="test-audience"
        )

    @pytest.mark.asyncio
    async def test_invalid_token_returns_none(self):
        """verify_token returns None when ScaleKit raises an exception."""
        mock_client = MagicMock()
        mock_client.validate_token.side_effect = Exception("invalid token")
        verifier = ScalekitTokenVerifier(mock_client)
        result = await verifier.verify_token("bad-token")
        assert result is None

    @pytest.mark.asyncio

        """verify_token handles missing sub claim gracefully."""
        mock_client = MagicMock()
        mock_client.validate_token.return_value = {"exp": 9999999999}
        verifier = ScalekitTokenVerifier(mock_client)
        result = await verifier.verify_token("token-no-sub")
        assert result is not None
        assert result.client_id == ""

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client -> /mcp (remote mode) | JWT required; validated by ScalekitTokenVerifier via FastMCP RequireAuthMiddleware |
| client -> /mcp (local mode) | Unauthenticated; localhost-only deployment |
| client -> /health | Always unauthenticated (custom_route exempt from auth) |
| server -> ScaleKit JWKS | TLS-protected; SDK fetches from {SCALEKIT_ENV_URL}/keys |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-19-01 | Spoofing | /mcp (remote mode) | mitigate | ScalekitTokenVerifier validates JWT signature via JWKS; returns None on any exception; FastMCP RequireAuthMiddleware returns 401 |
| T-19-02 | Spoofing | Token replay | mitigate | JWT exp claim checked by ScaleKit SDK; AccessToken.expires_at propagated to BearerAuthBackend |
| T-19-03 | Tampering | JWKS endpoint | mitigate | SDK fetches from fixed {SCALEKIT_ENV_URL}/keys over TLS; no user-controlled URL |
| T-19-04 | Denial of Service | Blocking event loop | mitigate | ScalekitClient.validate_token wrapped in asyncio.to_thread() — never blocks event loop |
| T-19-05 | Information Disclosure | Error messages | mitigate | ScalekitTokenVerifier catches all exceptions and returns None — no error details leaked to client |
| T-19-06 | Elevation of Privilege | Local mode bypass | accept | Local mode is intentionally unauthenticated per D-04; only enabled when SCALEKIT_* env vars are absent |
</threat_model>
