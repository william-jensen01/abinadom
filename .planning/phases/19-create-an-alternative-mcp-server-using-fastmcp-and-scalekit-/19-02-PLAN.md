def test_initialize_no_auth_remote_mode():

            })
    assert resp.status_code == 200


def test_tools_list_requires_auth_remote_mode():
    """AUTH-02: tools/list without token returns 401 when auth is enabled."""
    from fastmcp.server.auth.providers.debug import DebugTokenVerifier
    with patch("server._startup", return_value=_mock_startup()), \
         patch("server._auth", DebugTokenVerifier()):
        from server import mcp
        app = mcp.http_app()
        with TestClient(app, raise_server_exceptions=False) as client:
            # Initialize first to get session
            init_resp = client.post("/mcp", json={
                "jsonrpc": "2.0", "id": 1, "method": "initialize",
                "params": {"protocolVersion": "2025-06-18", "capabilities": {}, "clientInfo": {"name": "test", "version": "0"}},

            session_id = init_resp.headers.get("MCP-Session-Id", "")
            # tools/list without Authorization header
            resp = client.post("/mcp", json={
                "jsonrpc": "2.0", "id": 2, "method": "tools/list",

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client -> /mcp (remote mode) | Untrusted input; JWT required for tools/list and tools/call |
| client -> /mcp (local mode) | Localhost only; no auth by design |
| client -> /.well-known/* | Unauthenticated OAuth discovery endpoints (auto-served by ScalekitProvider) |
| server -> ScaleKit JWKS | Outbound HTTPS for key validation (trusted) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-19-05 | Spoofing | /mcp (remote mode) | mitigate | ScalekitProvider validates RS256 JWT via ScaleKit JWKS (`{env_url}/keys`); expired/invalid tokens return 401 |
| T-19-06 | Spoofing | /mcp (local mode) | accept | Local mode is intentionally unauthenticated; ALL THREE env vars must be set for remote mode — partial config stays local (D-04) |
| T-19-07 | Elevation of Privilege | initialize endpoint | mitigate | initialize is unauthenticated per MCP spec; tools/list and tools/call require auth in remote mode |
| T-19-08 | Information Disclosure | /.well-known/oauth-authorization-server | accept | OAuth discovery metadata is public by design (RFC 8414) |
</threat_model>
