```python
def build_talk_metadata_query(talk_id=None, conference_id=None, limit=None):
    query = """
        SELECT
            t.ID as talk_id, t.Title as title, t.Date as talk_date,
            t.Corpus as corpus, t.URL as url,
            t.URI_L as uri_l, t.URI_GC as uri_gc, t.data_aid as data_aid,
            t.SpeakerID as speaker_id, t.speaker_role_id,
            t.AudioUrl as audio_url, ...
            s.GivenNames as speaker_given_names, s.LastNames as speaker_last_names, ...
            c.ID as conference_id, c.Description as conference_description, ...
            cs.ID as session_id, cs.Description as session_description, ...
            ct.Sequence as talk_sequence, ...
        FROM talk t
        INNER JOIN speaker s ON t.SpeakerID = s.ID
        INNER JOIN conference_talk ct ON t.ID = ct.TalkID
        INNER JOIN conf_session cs ON ct.SessionID = cs.ID
        INNER JOIN conference c ON cs.ConferenceID = c.ID
        WHERE 1=1
    """
```
