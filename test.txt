      }
    }, ...}

title: gc.subitem.title OR gc.nav_item.title
subtitle: gc.nav_item.subtitle

Breakdown:
type: always "item"
src: gc.nav_item.image_renditions but it has to be formatted from
768x432,url
192x108,url
and just use the url
title: gc.subitem.title OR gc.nav_item.title
meta: gc.metadata.title formatted where month is abbreviated and remove "general conference"
secondaryMeta: gc.nav_item.subtitle,
subtype: "secondary"
